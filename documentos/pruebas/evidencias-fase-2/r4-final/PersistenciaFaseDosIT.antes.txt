                pedidos.actualizar(p);
            }
            return p;

    private <T> List<T> simultaneos(Callable<T> a,Callable<T> b) throws Exception {
        CountDownLatch listos=new CountDownLatch(2); CountDownLatch salida=new CountDownLatch(1);
        try(ExecutorService hilos=Executors.newFixedThreadPool(2)) {
            Callable<T> primero=() -> { listos.countDown(); if(!salida.await(10,TimeUnit.SECONDS)) throw new AssertionError("Sin salida concurrente"); return a.call(); };
            Callable<T> segundo=() -> { listos.countDown(); if(!salida.await(10,TimeUnit.SECONDS)) throw new AssertionError("Sin salida concurrente"); return b.call(); };
            Future<T> f1=hilos.submit(primero); Future<T> f2=hilos.submit(segundo);
            try { assertThat(listos.await(10,TimeUnit.SECONDS)).isTrue(); } finally { salida.countDown(); }
            return List.of(f1.get(30,TimeUnit.SECONDS),f2.get(30,TimeUnit.SECONDS));
        }
    }

    @Test void componenteDosAvancesSimultaneosConfirmanUnoYElOtroEtapaCambio() throws Exception {
        var p=prepararComponente(3); long v=leer(p.id()).version(); long n=eventos(p.id());
        Object resultado;
        try(ExecutorService hilos=Executors.newSingleThreadExecutor()) {
            var segundo=new java.util.concurrent.atomic.AtomicReference<Future<Object>>();
            tx(() -> {
                pedidos.buscarParaActualizar(p.id()).orElseThrow(); // Primera transacción conserva el bloqueo real.
                int bloqueador=jdbc.queryForObject("select pg_backend_pid()",Integer.class);
                segundo.set(hilos.submit(() -> {
                    try { accionComponente(p.id(),3,APROBAR,null,null); return "CONFIRMADO"; }
                    catch(RuntimeException error) { return error; } // Se afirma el tipo exacto al final; no se oculta.
                }));
                org.awaitility.Awaitility.await().atMost(java.time.Duration.ofSeconds(10)).until(() ->
                        Boolean.TRUE.equals(jdbc.queryForObject("select exists(select 1 from pg_stat_activity "
                                +"where ? = any(pg_blocking_pids(pid)))",Boolean.class,bloqueador)));
                accionComponente(p.id(),3,APROBAR,null,null);
                return null;
            }); // Commit de la primera; la segunda solo entonces obtiene el bloqueo.
            resultado=segundo.get().get(30,TimeUnit.SECONDS);
        }
        var leido=leer(p.id()); assertThat(leido.etapaActual()).isEqualTo(4); assertThat(leido.version()).isEqualTo(v+1);
        assertThat(eventos(p.id())).isEqualTo(n+1);
        assertThat(leido.etapas().stream().filter(e -> e.estado()==EstadoEtapa.ACTUAL).map(EtapaDePedido::numero)).containsExactly(4);
        assertThat(leido.etapas().get(2).estado()).isEqualTo(EstadoEtapa.COMPLETADA);
        assertThat(leido.etapas().get(3).reentradas()).isZero();
        System.out.println("B2: bloqueo PostgreSQL observado; un avance confirmado; un evento nuevo; una etapa ACTUAL; segunda respuesta: "+resultado);
        assertThat(resultado).isInstanceOfSatisfying(EtapaCambio.class,error -> assertThat(error.codigo()).isEqualTo(CodigoError.ETAPA_CAMBIO));
    }
